13: Bit 0 - BG Display(for CGB see below) (0 = Off, 1 = On)

79: void GPU::changeModeGPU(MMU* mmu, uint8_t gpuMode, Interrupt* interr) {
80: 	mode = gpuMode;
81: 
82: 	// 1. Leemos el valor actual del registro STAT
83: 	uint8_t lcdStatValue = mmu->read8(0xFF41);
84: 
85: 	// 2. Limpiamos los bits 0 y 1 (Modo antiguo)
86: 	lcdStatValue &= 0xFC;
87: 
88: 	// 3. Aplicamos el nuevo modo al VALOR (no a la direcci¢n)
89: 	lcdStatValue |= (mode & 0x03);
90: 
91: 	// 4. Escribimos de vuelta en memoria
92: 	mmu->io[0x41] = lcdStatValue;
93: 
94: 	// 5. Manejo de Interrupciones STAT (Opcional pero recomendado aqu¡)
95: 	// Si el modo nuevo coincide con la interrupci¢n seleccionada en STAT, solicitarla.
96: 	bool interruptTriggered = false;
97: 	if ((mode == 0) && (lcdStatValue & 0x08)) interruptTriggered = true; // Mode 0 HBlank check
98: 	if ((mode == 1) && (lcdStatValue & 0x10)) interruptTriggered = true; // Mode 1 VBlank check
99: 	if ((mode == 2) && (lcdStatValue & 0x20)) interruptTriggered = true; // Mode 2 OAM check
100: 	if (interruptTriggered) {
101: 		interr->requestInterrupt(mmu, 1);
102: 	}
103: }

119: uint8_t GPU::getColour(uint8_t colourNum, uint16_t address, MMU *mmu) {
120: 	uint8_t res = 0; //Full white
121: 	uint8_t palette = mmu->read8(address);
122: 	uint8_t hi = 0;
123: 	uint8_t lo = 0;
124: 
125: 	// which bits of the colour palette does the colour id map to?
126: 	switch (colourNum)
127: 	{
128: 	case 0: hi = 1; lo = 0; break;
129: 	case 1: hi = 3; lo = 2; break;
130: 	case 2: hi = 5; lo = 4; break;
131: 	case 3: hi = 7; lo = 6; break;
132: 	}
133: 
134: 	// use the palette to get the colour
135: 	uint8_t colour = 0;
136: 	colour = BitGetVal(palette, hi) << 1;
137: 	colour |= BitGetVal(palette, lo);
138: 
139: 	// convert the game colour to emulator colour
140: 
141: 	res = colour;
142: 
143: 	return res;
144: 
145: }

147: void GPU::cgbPaletteToRGB(const uint8_t* palRAM, int paletteNum, int colorIdx, uint8_t& r, uint8_t& g, uint8_t& b) {
148: 
149:     int idx = (paletteNum * 8) + (colorIdx * 2);
150: 
151:     uint16_t rgb555 = palRAM[idx] | ((uint16_t)palRAM[idx + 1] << 8);
152: 
153:     r = (uint8_t)(((rgb555 >> 0) & 0x1F) * 255 / 31);
154: 
155:     g = (uint8_t)(((rgb555 >> 5) & 0x1F) * 255 / 31);
156: 
157:     b = (uint8_t)(((rgb555 >> 10) & 0x1F) * 255 / 31);
158: 
159: }

179: void GPU::renderBackground(MMU* mmu) {
180: 	uint8_t lcdControl = mmu->read8(0xFF40);
181: 	uint8_t ly = mmu->read8(0xFF44); // L¡nea actual (Scanline)
182: 
183: 	// Coordenadas de Scroll y Window
184: 	uint8_t scrollY = mmu->read8(0xFF42);
185: 	uint8_t scrollX = mmu->read8(0xFF43);
186: 	uint8_t windowY = mmu->read8(0xFF4A);
187: 	uint8_t windowX = mmu->read8(0xFF4B) - 7; // WX tiene un offset de 7
188: 
189: 	// Flags del LCDC
190: 	bool windowEnabled = isKthBitSet(lcdControl, 5);
191: 	bool tileDataUnsigned = isKthBitSet(lcdControl, 4); // Bit 4: 1=8000-8FFF, 0=8800-97FF
192: 	bool bgMapSelect = isKthBitSet(lcdControl, 3);      // Bit 3: Mapa Fondo (9800 vs 9C00)
193: 	bool winMapSelect = isKthBitSet(lcdControl, 6);     // Bit 6: Mapa Ventana (9800 vs 9C00)
194: 
195: 	// Direcciones base de los mapas
196: 	uint16_t bgMapBase = bgMapSelect ? 0x9C00 : 0x9800;
197: 	uint16_t winMapBase = winMapSelect ? 0x9C00 : 0x9800;
198: 
199: 	bool windowDrawnThisLine = false;
200: 	// --- BUCLE DE PIXELES (0 a 159) ---
201: 	for (int pixel = 0; pixel < 160; pixel++) {
202: 
203: 		// 1. Decidir si dibujamos VENTANA o FONDO en este p¡xel espec¡fico
204: 		bool usingWindow = false;
205: 
206: 		if (windowEnabled) {
207: 			// La ventana se dibuja si estamos dentro de su rango Y y X
208: 			if (ly >= windowY && pixel >= windowX) {
209: 				usingWindow = true;
210: 				windowDrawnThisLine = true;
211: 			}
212: 		}
213: 
214: 		// 2. Calcular las coordenadas en el mapa de tiles (VRAM)
215: 		uint16_t mapBase = 0;
216: 		uint8_t yPos = 0;
217: 		uint8_t xPos = 0;
218: 
219: 		if (usingWindow) {
220: 			mapBase = winMapBase;
221: 			yPos = wly;       // Y relativo a la ventana
222: 			xPos = pixel - windowX;    // X relativo a la ventana
223: 		}
224: 		else {
225: 			mapBase = bgMapBase;
226: 			yPos = scrollY + ly;       // Y relativo al fondo (con scroll)
227: 			xPos = scrollX + pixel;    // X relativo al fondo (con scroll)
228: 		}
229: 
230: 		// 3. Obtener el ID del Tile
231: 		// (yPos / 8) * 32  -> Fila del tile (32 tiles de ancho)
232: 		// (xPos / 8)       -> Columna del tile
233: 		uint16_t tileRow = (yPos / 8) * 32;
234: 		uint16_t tileCol = (xPos / 8);
235: 		uint16_t tileAddress = mapBase + tileRow + tileCol;
236: 
237: 		// Leemos el indice del tile (puede ser signed o unsigned) 		uint8_t tileIndex = mmu->read8(tileAddress);  		// CGB: tile attributes from VRAM bank 1 		uint8_t tileAttr = 0; 		if (mmu->cgbMode) 			tileAttr = mmu->vramBank[1][tileAddress - 0x8000]; 		bool cgbBgPriority = (tileAttr & 0x80) != 0; 		bool cgbYFlip      = (tileAttr & 0x40) != 0; 		bool cgbXFlip      = (tileAttr & 0x20) != 0; 		int  cgbVramBank   = (tileAttr & 0x08) ? 1 : 0; 		int  cgbPalette    = (tileAttr & 0x07);  		// 4. Calcular la direccion de los datos del tile 		uint16_t tileLocation = 0;  		if (tileDataUnsigned) { 			tileLocation = 0x8000 + (tileIndex * 16); 		} 		else { 			tileLocation = 0x9000 + ((int8_t)tileIndex * 16); 		}  		// 5. Obtener los bytes del tile 		uint8_t tileLineOfs = (yPos % 8); 		if (cgbYFlip) tileLineOfs = 7 - tileLineOfs; 		uint8_t tileLine = tileLineOfs * 2; 		uint8_t data1, data2; 		if (mmu->cgbMode && cgbVramBank == 1) { 			data1 = mmu->vramBank[1][(tileLocation - 0x8000) + tileLine]; 			data2 = mmu->vramBank[1][(tileLocation - 0x8000) + tileLine + 1]; 		} else { 			data1 = mmu->read8(tileLocation + tileLine); 			data2 = mmu->read8(tileLocation + tileLine + 1); 		}  		// 6. Decodificar el color 		int colorBit = cgbXFlip ? (xPos % 8) : (7 - (xPos % 8)); 		uint8_t colorNum = !!(data2 & (1 << colorBit)) << 1; 		colorNum |= !!(data1 & (1 << colorBit));  		// 7. Colorear 		uint8_t red = 0, green = 0, blue = 0; 		if (mmu->cgbMode) { 			cgbPaletteToRGB(mmu->bgPaletteRAM, cgbPalette, colorNum, red, green, blue); 		} else { 			uint8_t col = getColour(colorNum, 0xFF47, mmu); 			red   = GB_PALETTES[currentPalette].color[col][0]; 			green = GB_PALETTES[currentPalette].color[col][1]; 			blue  = GB_PALETTES[currentPalette].color[col][2]; 		}  		// Escribir al framebuffer 		if (ly < 144 && pixel < 160) { 			bgColorIndex[pixel] = colorNum; 			framebuffer[ly][pixel][0] = red; 			framebuffer[ly][pixel][1] = green; 			framebuffer[ly][pixel][2] = blue; 		}
238: 	}
239: 	if (windowDrawnThisLine) wly++;
240: }

242: void GPU::renderSprites(MMU *mmu) {  	bool use8x16 = isKthBitSet(mmu->read8(0xFF40), 2); 	int ysize = use8x16 ? 16 : 8; 	int scanline = mmu->read8(0xFF44);  	for (int sprite = 39; sprite >= 0; sprite--) 	{ 		uint8_t index      = sprite * 4; 		int     yPos       = (int)mmu->read8(0xFE00 + index)     - 16; 		int     xPos       = (int)mmu->read8(0xFE00 + index + 1) - 8; 		uint8_t tileLoc    = mmu->read8(0xFE00 + index + 2); 		uint8_t attributes = mmu->read8(0xFE00 + index + 3);  		bool yFlip    = isKthBitSet(attributes, 6); 		bool xFlip    = isKthBitSet(attributes, 5); 		bool priority = isKthBitSet(attributes, 7);  		if (scanline < yPos || scanline >= yPos + ysize) continue;  		if (use8x16) tileLoc &= 0xFE;  		int line = scanline - yPos; 		if (yFlip) line = (ysize - 1) - line;  		int  cgbSpVramBank = mmu->cgbMode ? ((attributes & 0x08) ? 1 : 0) : 0; 		int  cgbSpPalette  = mmu->cgbMode ? (attributes & 0x07) : 0;  		uint16_t dataAddress = 0x8000 + (tileLoc * 16) + (line * 2); 		uint8_t  data1, data2; 		if (mmu->cgbMode && cgbSpVramBank == 1) { 			data1 = mmu->vramBank[1][(dataAddress - 0x8000)]; 			data2 = mmu->vramBank[1][(dataAddress - 0x8000) + 1]; 		} else { 			data1 = mmu->read8(dataAddress); 			data2 = mmu->read8(dataAddress + 1); 		}  		uint16_t paletteAddr = isKthBitSet(attributes, 4) ? 0xFF49 : 0xFF48;  		for (int tilePixel = 7; tilePixel >= 0; tilePixel--) 		{ 			int colourbit = xFlip ? (7 - tilePixel) : tilePixel; 			int colourNum = (BitGetVal(data2, colourbit) << 1) | BitGetVal(data1, colourbit); 			if (colourNum == 0) continue; 			int pixel = xPos + (7 - tilePixel); 			if (scanline < 0 || scanline > 143 || pixel < 0 || pixel > 159) continue; 			if (priority && bgColorIndex[pixel] != 0) continue; 			uint8_t red = 0, green = 0, blue = 0; 			if (mmu->cgbMode) { 				cgbPaletteToRGB(mmu->objPaletteRAM, cgbSpPalette, colourNum, red, green, blue); 			} else { 				uint8_t col = getColour(colourNum, paletteAddr, mmu); 				red   = GB_PALETTES[currentPalette].color[col][0]; 				green = GB_PALETTES[currentPalette].color[col][1]; 				blue  = GB_PALETTES[currentPalette].color[col][2]; 			} 			framebuffer[scanline][pixel][0] = red; 			framebuffer[scanline][pixel][1] = green; 			framebuffer[scanline][pixel][2] = blue; 		} 	} }
243: void GPU::step(uint16_t cycles, MMU *mmu, SDL_Renderer *render, Interrupt *interr) {
244: 	// 1. LEER EL REGISTRO DE CONTROL (LCDC)
245: 	uint8_t lcdc = mmu->io[0x40];
246: 
247: 	// 2. COMPROBAR EL BIT 7 (LCD ENABLE)
248: 	bool isLCDEnabled = (lcdc & 0x80); // 0x80 es 10000000 en binario
249: 
250: 	if (!isLCDEnabled) {
251: 		// --- LA PANTALLA ESTµ APAGADA ---
252: 
253: 		// El hardware real resetea LY a 0 cuando se apaga el LCD
254: 		clock = 0;      // Reseteamos el contador de ciclos internos
255: 		line = 0;       // LY vuelve a 0
256: 		mode = 1;       // Modo HBlank (o estado inicial)
257: 
258: 		// Escribimos el estado en memoria para que la CPU lo lea correctamente
259: 		mmu->io[0x44] = 0; // LY = 0
260: 
261: 		// Importante: Tambi‚n se resetean los bits de modo en STAT (bits 0-1)
262: 		uint8_t stat = mmu->io[0x41];
263: 		stat &= 0xFC; // Borrar bits 0 y 1
264: 		mmu->io[0x41] = stat;
265: 
266: 		return; // ­SALIR! No procesar nada m s mientras est‚ apagada
267: 	}
268: 
269: 	clock += cycles;
270: 	switch (mode)
271: 		{
272: 		case 0:
273: 			//Horizontal Blanking
274: 			//std::cout << "Entering Horizontal Blanking" << std::endl;
275: 			if (clock >= 204) {
276: 				clock -= 204;
277: 				line++;
278: 				mmu->io[0x44] = line;
279: 				checkLYC = false;
280: 
281: 				if (line == 144) {
282: 					//Enter in Vertical Blanking Mode
283: 					changeModeGPU(mmu, 1, interr);
284: 					interr->requestInterrupt(mmu, 0);
285: 
286: 					//TODO: Write a function that write data into the SDL Render
287: 				// renderFramebuffer moved to main loop
288: 					//std::cout << "Writing data from framebuffer" << std::endl;
289: 				}
290: 				else {
291: 					changeModeGPU(mmu, 2, interr);
292: 				}
293: 			}
294: 			break;
295: 		case 1:
296: 			//Vertical Blanking
297: 			//std::cout << "Entering Vertical Blanking" << std::endl;
298: 			if (clock >= 456) {
299: 
300: 				clock -= 456;
301: 				line++;
302: 
303: 				if (line > 153) {
304: 				line = 0;
305: 				wly = 0;
306: 				checkLYC = false;
307: 				mmu->io[0x44] = 0;
308: 				changeModeGPU(mmu, 2, interr);
309: 			} else {
310: 				checkLYC = false;
311: 				mmu->io[0x44] = line;
312: 			}
313: 			}
314: 			break;
315: 		case 2:
316: 			//Read OAM
317: 			//std::cout << "Reading OAM Mode" << std::endl;
318: 			if (clock >= 80) {
319: 				clock -= 80;
320: 				changeModeGPU(mmu, 3, interr);
321: 			}
322: 			break;
323: 		case 3:
324: 			//Read VRAM for generate picture
325: 			//std::cout << "Reading VRAM for generate picture" << std::endl;
326: 			if (clock >= 172) {
327: 
328: 				// Draw scanline with registers valid during Mode 3, THEN enter HBlank
329: 				clock -= 172;
330: 				DrawScanline(mmu);
331: 				changeModeGPU(mmu, 0, interr); // HBlank STAT interrupt fires after draw
332: 			}
333: 			break;
334: 		}
335: 	
336: 
337: 	//Comparaci¢n LYC(Coincidence Flag)
338: 	uint8_t currentLY = mmu->io[0x44];
339: 	uint8_t currentLYC = mmu->io[0x45];
340: 
341: 	// La Game Boy comprueba si LY == LYC y lanza interrupci¢n si est  habilitada
342: 	if (currentLY == currentLYC) {
343: 		mmu->io[0x41] |= 0x04; // Set Coincidence Bit (Bit 2)
344: 
345: 		// Solo pedimos la interrupci¢n si NO la hab¡amos pedido ya para esta coincidencia
346: 		if (checkLYC == false) {
347: 			// Comprobar si la interrupci¢n STAT por LYC est  habilitada (Bit 6 de STAT)
348: 			if (mmu->io[0x41] & 0x40) {
349: 				interr->requestInterrupt(mmu, 1);
350: 			}
351: 			checkLYC = true; // Marcamos que ya hemos gestionado esta coincidencia
352: 		}
353: 	}
354: 	else {
355: 		mmu->io[0x41] &= ~0x04; // Reset Coincidence Bit
356: 		checkLYC = false; // Reseteamos el flag para la pr¢xima vez que coincidan
357: 	}
358: }
